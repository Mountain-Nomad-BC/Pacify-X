def https_origin(origin, addresses):
    parsed = urlsplit(origin)
    if (parsed.scheme != 'https' or not parsed.hostname or parsed.port not in (None, 443)
            or parsed.username is not None or parsed.password is not None or parsed.path
            or parsed.query or parsed.fragment or origin != 'https://' + parsed.hostname
            or re.fullmatch(r'[a-z0-9](?:[a-z0-9.-]{0,251}[a-z0-9])?', parsed.hostname) is None
            or '.' not in parsed.hostname or parsed.hostname.endswith(('.localhost', '.local'))):
        raise ValueError('exact public HTTPS origin required')
    if type(addresses) is not list or not 1 <= len(addresses) <= 8 or len(set(addresses)) != len(addresses):
        raise ValueError('explicit approved address pins required')
    for address in addresses:
        if type(address) is not str or not ipaddress.ip_address(address).is_global:
            raise ValueError('private/non-global endpoint address refused')
    return parsed.hostname
